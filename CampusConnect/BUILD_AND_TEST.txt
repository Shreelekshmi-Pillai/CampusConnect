   CAMPUS_CONNECT_DB_PASSWORD = your MySQL password
   Optional: CAMPUS_CONNECT_DB_URL = jdbc:mysql://localhost:3306/campus_connect?serverTimezone=Asia/Kolkata&useSSL=false&allowPublicKeyRetrieval=true
